# Retail POS pod — Quadlet unit
#
# Install (rootless, per-user systemd):
#   mkdir -p ~/.config/containers/systemd
#   cp deploy/quadlet/* ~/.config/containers/systemd/
#   systemctl --user daemon-reload
#   systemctl --user enable --now retail-pos-pod.service retail-pos-backend.service
#
# Rootful hosts: copy to /etc/containers/systemd/ and use `sudo systemctl` without
# `--user`. See deploy/PRODUCTION-DEPLOYMENT.md.
#
# The pod only creates the network namespace and the port forwards. The
# containers in this directory attach to it.

[Unit]
Description=Retail POS pod (retail-pos-system)
# This file becomes retail-pos-pod.service; the containers reference it with
# Pod=retail-pos.pod. Quadlet links them automatically.

[Pod]
# Without this, Quadlet would name the pod systemd-retail-pos.
PodName=retail-pos-pod
Network=bridge
# 8080 and 5432 bind to 127.0.0.1 only. An unqualified PublishPort reaches every
# host interface, which would put the unauthenticated API and the database in
# reach of anything that can route to this host. The backend reaches postgres
# over the pod's own network namespace, so no host port is needed for the
# running system; the loopback listener exists for `podman-deploy.sh seed`,
# which connects from the host.
PublishPort=127.0.0.1:8080:8080
# nginx listens on 8081 inside the frontend image (deploy/nginx/default.conf.template,
# EXPOSE 8081), not 80. The retired systemd unit published 80:80 and could
# never have worked.
#
# 8000 stays on every interface: a browser on the store LAN is the intended
# client. Put a reverse proxy with TLS in front of it for anything off-LAN.
# 5173 is deliberately avoided — that is the Vite dev server's port, so a
# developer running `npm run dev` alongside this stack would collide with it.
PublishPort=8000:8081
PublishPort=127.0.0.1:5432:5432
