# Auth Flow Implementation Plan

## Current State Analysis

### Backend (Go/Gin)
- **Existing**: Login/logout handlers, `/api/auth/validate` endpoint
- **Issue**: No `/api/refresh` endpoint despite refresh token stored in DB
- **Cookie**: Access token set via HttpOnly `session_token` cookie (expires 24h)
- **Refresh token**: Stored in DB with 7d expiry, but only `authService.RefreshToken()` exists with no HTTP endpoint

### Frontend (SvelteKit)
- **Existing**: Single layout `+layout.svelte`, hash-based SPA routing, stores use localStorage
- **Issues**: 
  - localStorage stores user data (security violation)
  - No route groups separation
  - No token refresh flow
  - Single layout renders sidebar conditionally (not proper separation)

func (h *Handler) RefreshToken(c *gin.Context) {
	var input struct {
		RefreshToken string `json:"refresh_token"`
	}
	if err := c.ShouldBindJSON(&input); err != nil {
		// Also accept from header
		refreshToken := c.GetHeader("X-Refresh-Token")
		if refreshToken == "" {
			c.JSON(http.StatusBadRequest, gin.H{"error": "Refresh token required"})
			return
		}
		input.RefreshToken = refreshToken
	}

	tokenPair, err := h.authService.RefreshToken(c.Request.Context(), input.RefreshToken)
	if err != nil {
		c.JSON(http.StatusUnauthorized, gin.H{"error": "Invalid refresh token"})
		return
	}

	// Set new access token cookie
	http.SetCookie(c.Writer, &http.Cookie{
		Name:     "session_token",
		Value:    tokenPair.AccessToken,
		Path:     "/",
		HttpOnly: true,
		SameSite: http.SameSiteStrictMode,
		Secure:   true,
		Expires:  time.Now().Add(24 * time.Hour),
	})

	c.JSON(http.StatusOK, gin.H{
		"message":       "Token refreshed",
		"refresh_token": tokenPair.RefreshToken, // Return new refresh token
	})
}
```

package auth

import (
	"net/http"
	"strings"

	"github.com/gin-gonic/gin"
)

func RoleMiddleware(allowedRoles ...string) gin.HandlerFunc {
	return func(c *gin.Context) {
		role := c.GetString("role")
		
		for _, allowed := range allowedRoles {
			if role == allowed {
				c.Next()
				return
			}
		}
		
		c.JSON(http.StatusForbidden, gin.H{"error": "Access denied"})
		c.Abort()
	}
}

func CashierOnly() gin.HandlerFunc {
	return RoleMiddleware("admin", "cashier")
}
```

### Phase 3: Auth Store (No localStorage)

**File**: `src/lib/stores/auth.ts` (new file)

```typescript
import { writable, get } from 'svelte/store';

export interface User {
    id: number;
    username: string;
    role: string;
}

interface AuthState {
    user: User | null;
    isAuthenticated: boolean;
    loading: boolean;
}

function createAuthStore() {
    const { subscribe, set, update } = writable<AuthState>({
        user: null,
        isAuthenticated: false,
        loading: true
    });

    return {
        subscribe,
        setUser: (user: User | null) => update(s => ({ ...s, user, isAuthenticated: !!user, loading: false })),
        setLoading: (loading: boolean) => update(s => ({ ...s, loading })),
        reset: () => set({ user: null, isAuthenticated: false, loading: false }),
        getUser: () => get({ subscribe }).user
    };
}

export const auth = createAuthStore();
```

### Phase 4: Fetch Wrapper with Token Refresh

**File**: `src/lib/api/client.ts` (new file)

```typescript
import axios from 'axios';
import { auth } from '$lib/stores/auth';
import { goto } from '$app/navigation';

const client = axios.create({
    baseURL: '/api',
    withCredentials: true,
    timeout: 10000
});

let isRefreshing = false;

client.interceptors.response.use(
    res => res,
    async error => {
        const originalRequest = error.config;
        
        if (error.response?.status === 401 && !originalRequest._retry) {
            originalRequest._retry = true;
            
            if (isRefreshing) {
                return Promise.reject(error);
            }
            
            isRefreshing = true;
            
            try {
                // Get stored refresh token (from memory, NOT localStorage)
                const refreshToken = sessionStorage.getItem('refresh_token');
                
                if (!refreshToken) {
                    throw new Error('No refresh token');
                }
                
                await client.post('/refresh', {}, {
                    headers: { 'X-Refresh-Token': refreshToken }
                });
                
                // Retry original request
                return client(originalRequest);
            } catch (refreshError) {
                auth.reset();
                sessionStorage.removeItem('refresh_token');
                goto('/login');
                return Promise.reject(refreshError);
            } finally {
                isRefreshing = false;
            }
        }
        
        return Promise.reject(error);
    }
);

export default client;
```

      // Login - backend sets cookie

      goto('/dashboard');

      error = e.response?.data?.error || 'Login failed';
    } finally {
      loading = false;
    }
  }

import client from './client';
import { auth } from '$lib/stores/auth';
import { goto } from '$app/navigation';

export async function checkAuth() {
    try {
        const { data } = await client.get('/auth/validate');
        auth.setUser(data.user);
        return true;
    } catch {
        auth.reset();
        return false;
    }
}

export async function logout() {
    try {
        await client.post('/logout');
    } catch {
        // Ignore errors
    }
    sessionStorage.removeItem('refresh_token');
    auth.reset();
    goto('/login');
}

### Frontend Route Guard with Role Check

**File**: `src/routes/hooks.client.ts` (new)

```typescript
import { checkAuth } from '$lib/api/auth';
import { get } from 'svelte/store';
import { auth } from '$lib/stores/auth';

const ROLE_PERMISSIONS: Record<string, string[]> = {
	admin: ['/dashboard', '/inventory', '/reports', '/pos'],
	cashier: ['/pos']
};

export async function handle({ event, resolve }) {
	const path = event.url.pathname;
	
	// Allow login page without auth check
	if (path === '/login') {
		return resolve(event);
	}
	
	// Check authentication
	const isAuth = await checkAuth();
	const state = get(auth);
	
	if (!isAuth || !state.user) {
		return Response.redirect('/login', 303);
	}
	
	// Role-based route protection
	const allowedRoutes = ROLE_PERMISSIONS[state.user.role] || ['/pos'];
	const hasAccess = allowedRoutes.some(route => path.startsWith(route));
	
	if (!hasAccess) {
		// Redirect to allowed page based on role
		return Response.redirect(allowedRoutes[0] || '/pos', 303);
	}
	
	return resolve(event);
}
```
